cmake_minimum_required(VERSION 3.16)
project(team02_race)
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
find_package(ament_index_cpp REQUIRED)
find_package(gz-sim8 REQUIRED)
find_package(gz-transport13 REQUIRED)
find_package(gz-msgs10 REQUIRED)
ament_python_install_package(team02_race)
add_library(team02_pose_02 SHARED pose_plugin/Team02Pose.cc)
target_compile_features(team02_pose_02 PRIVATE cxx_std_17)
ament_target_dependencies(team02_pose_02 ament_index_cpp)
target_link_libraries(team02_pose_02 gz-sim8::gz-sim8 gz-transport13::gz-transport13 gz-msgs10::gz-msgs10)
add_library(team02_battle_replay_02 SHARED legacy/BattleReplay08.cc)
target_compile_features(team02_battle_replay_02 PRIVATE cxx_std_17)
target_link_libraries(team02_battle_replay_02 PRIVATE gz-sim8::gz-sim8 gz-transport13::gz-transport13 gz-msgs10::gz-msgs10)
install(TARGETS team02_pose_02 team02_battle_replay_02 LIBRARY DESTINATION lib)
install(PROGRAMS scripts/team02_control DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch maps models DESTINATION share/${PROJECT_NAME})
install(FILES race_entry.json DESTINATION share/${PROJECT_NAME})
ament_package()
