cmake_minimum_required(VERSION 3.16)
project(team08_race)
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
find_package(ament_index_cpp REQUIRED)
find_package(gz-sim8 REQUIRED)
find_package(gz-transport13 REQUIRED)
find_package(gz-msgs10 REQUIRED)
ament_python_install_package(team08_race)
add_library(team08_pose_08 SHARED pose_plugin/Team08Pose.cc)
target_compile_features(team08_pose_08 PRIVATE cxx_std_17)
ament_target_dependencies(team08_pose_08 ament_index_cpp)
target_link_libraries(team08_pose_08 gz-sim8::gz-sim8 gz-transport13::gz-transport13 gz-msgs10::gz-msgs10)
add_library(team08_battle_replay_08 SHARED legacy/BattleReplay08.cc)
target_compile_features(team08_battle_replay_08 PRIVATE cxx_std_17)
target_link_libraries(team08_battle_replay_08 PRIVATE gz-sim8::gz-sim8 gz-transport13::gz-transport13 gz-msgs10::gz-msgs10)
install(TARGETS team08_pose_08 team08_battle_replay_08 LIBRARY DESTINATION lib)
install(PROGRAMS scripts/team08_control DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch maps models DESTINATION share/${PROJECT_NAME})
install(FILES race_entry.json DESTINATION share/${PROJECT_NAME})
ament_package()
