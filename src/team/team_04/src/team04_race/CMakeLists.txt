cmake_minimum_required(VERSION 3.16)
project(team04_race)
set(CMAKE_CXX_STANDARD 17)
option(TEAM04_OPTIMIZE_PLANNER "Optimize LiDAR node without fast-math or physics changes" ON)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
foreach(dep rclcpp sensor_msgs std_msgs geometry_msgs nav_msgs visualization_msgs gz_sim_vendor gz_plugin_vendor gz_transport_vendor gz_msgs_vendor)
  find_package(${dep} REQUIRED)
endforeach()
find_package(gz-sim8 REQUIRED)
find_package(gz-plugin2 REQUIRED COMPONENTS register)
find_package(gz-transport13 REQUIRED)
find_package(gz-msgs10 REQUIRED)
add_library(team04_vehicle_system SHARED src/vehicle_system.cpp)
target_link_libraries(team04_vehicle_system gz-sim8::gz-sim8 gz-plugin2::register gz-transport13::core gz-msgs10::core)
add_library(team04_traffic_actor SHARED tests/traffic_actor_system.cpp)
target_link_libraries(team04_traffic_actor gz-sim8::gz-sim8 gz-plugin2::register gz-transport13::core gz-msgs10::core)
install(TARGETS team04_traffic_actor LIBRARY DESTINATION lib)
add_library(team04_trial_observer SHARED tests/trial_observer.cpp)
target_link_libraries(team04_trial_observer gz-sim8::gz-sim8 gz-plugin2::register gz-transport13::core gz-msgs10::core)
install(TARGETS team04_trial_observer LIBRARY DESTINATION lib)
add_executable(control_04 src/control.cpp)
ament_target_dependencies(control_04 rclcpp std_msgs sensor_msgs geometry_msgs)
add_executable(scan_04 src/scan.cpp)
ament_target_dependencies(scan_04 rclcpp sensor_msgs)
set(TEAM04_TRAFFIC_SOURCES src/lidar_motion.cpp src/vehicle_detector.cpp src/vehicle_tracker.cpp src/collision_predictor.cpp src/overtake_planner.cpp)
add_executable(autonomy_04 src/autonomy.cpp ${TEAM04_TRAFFIC_SOURCES} src/lidar_planner.cpp src/centerline_planner.cpp src/observed_corridor.cpp)
target_include_directories(autonomy_04 PRIVATE include)
if(TEAM04_OPTIMIZE_PLANNER)
  target_compile_options(autonomy_04 PRIVATE -O2)
endif()
ament_target_dependencies(autonomy_04 rclcpp sensor_msgs std_msgs geometry_msgs nav_msgs visualization_msgs)
install(TARGETS team04_vehicle_system LIBRARY DESTINATION lib)
install(TARGETS control_04 scan_04 autonomy_04 DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config models meshes docs DESTINATION share/${PROJECT_NAME})
install(FILES race_entry.json README.md DESTINATION share/${PROJECT_NAME})
install(PROGRAMS scripts/validate_dimensions.py scripts/generate_model.py DESTINATION share/${PROJECT_NAME}/scripts)
ament_environment_hooks(env-hooks/team04.dsv.in)
if(BUILD_TESTING)
  enable_testing()
  add_executable(team04_planner_tests tests/planner_tests.cpp src/lidar_planner.cpp src/centerline_planner.cpp src/observed_corridor.cpp)
  target_include_directories(team04_planner_tests PRIVATE include)
  if(TEAM04_OPTIMIZE_PLANNER)
    target_compile_options(team04_planner_tests PRIVATE -O2)
  endif()
  add_test(NAME team04_planner_tests COMMAND team04_planner_tests)
  add_executable(team04_racing_tests tests/racing_planner_tests.cpp src/lidar_planner.cpp src/centerline_planner.cpp src/observed_corridor.cpp)
  target_include_directories(team04_racing_tests PRIVATE include)
  target_compile_options(team04_racing_tests PRIVATE -O2)
  add_test(NAME team04_racing_tests COMMAND team04_racing_tests)
  add_executable(team04_traffic_tests tests/traffic_tests.cpp ${TEAM04_TRAFFIC_SOURCES} src/lidar_planner.cpp src/centerline_planner.cpp src/observed_corridor.cpp)
  target_include_directories(team04_traffic_tests PRIVATE include)
  target_compile_options(team04_traffic_tests PRIVATE -O2)
  add_test(NAME team04_traffic_tests COMMAND team04_traffic_tests)
endif()
ament_package()
